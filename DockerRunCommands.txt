###################################################### ПОДГОТОВКА ######################################################

# создаём свою сеть:
sudo docker network create sportnotifier_network
# посмотреть какие контейнеры находятся в сети:
sudo docker network inspect sportnotifier_network

# установка postgres (вставь пароль сразу какой требуется, чтобы потом не переделывать):
# ВНИМАНИЕ:
#   приложениям внутри сети sportnotifier_network подключаться к БД по внутреннему порту docker, а не к пробрасываемому! То есть если 5444:5432, то конектимся к 5432
#   приложениям из вне, к пробрасываемому, если брать пример выше, то это будет: 5444
sudo docker run -d --network sportnotifier_network --restart always --name postgres -e POSTGRES_PASSWORD=password -p 5435:5432 postgres:16.0-alpine3.18
# либо docker_postgres_create.sh
