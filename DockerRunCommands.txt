# КОМАНДЫ:

#билдим проект:
./gradlew clean build

#собираем образ:
docker build -t slezkin23/sportnotifier:latest .

#запускаем контейнер:
docker run -it --name sportnotifier -p 8090:8090 --env-file Dockerfile_envs.env -d slezkin23/sportnotifier

# RUN - ALL: всех команд перечисленных выше за 1 раз:
./gradlew clean build && docker build -t slezkin23/sportnotifier:latest . && docker run -it --name sportnotifier -p 8090:8090 --env-file Dockerfile_envs.env -d slezkin23/sportnotifier

# запушить образ в hub !!! ток сначала обнови переменные в application.yml
docker push slezkin23/sportnotifier:latest
# скачать
docker pull slezkin23/sportnotifier:latest && docker run -it --name sportnotifier -p 8090:8090 -d slezkin23/sportnotifier && docker logs -f sportnotifier
